// Center-Crop Engine (Distortion / Stretch Fix) function captureFromFrame(targetWidth, targetHeight, isBlackAndWhite) { const canvas = document.createElement('canvas'); canvas.width = targetWidth; canvas.height = targetHeight; const ctx = canvas.getContext('2d'); // Camera Video Stream Ki Actual Aspect Ratio Calculation let videoWidth = video.videoWidth || 1280; let videoHeight = video.videoHeight || 720; let targetRatio = targetWidth / targetHeight; let videoRatio = videoWidth / videoHeight; let sourceWidth, sourceHeight, startX, startY; // Calculate crop coordinates without distorting/stretching photo if (videoRatio > targetRatio) { sourceHeight = videoHeight; sourceWidth = videoHeight * targetRatio; startX = (videoWidth - sourceWidth) / 2; startY = 0; } else { sourceWidth = videoWidth; sourceHeight = videoWidth / targetRatio; startX = 0; startY = (videoHeight - sourceHeight) / 2; } // Exact center crop drawn to target canvas ctx.drawImage( video, startX, startY, sourceWidth, sourceHeight, // Crop area from live camera 0, 0, targetWidth, targetHeight // Drawn size on frame ); // Apply B&W + High Contrast ONLY if Signature step if (isBlackAndWhite) { const imgData = ctx.getImageData(0, 0, targetWidth, targetHeight); const data = imgData.data; for (let i = 0; i < data.length; i += 4) { let avg = (data[i] + data[i + 1] + data[i + 2]) / 3; // High contrast background cleanup if (avg > 130) { data[i] = 255; data[i + 1] = 255; data[i + 2] = 255; } else { let darkVal = avg * 0.6; data[i] = darkVal; data[i + 1] = darkVal; data[i + 2] = darkVal; } } ctx.putImageData(imgData, 0, 0); } return canvas; }
